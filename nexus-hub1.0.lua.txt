local function Create(class, props, parent)
    local obj = Instance.new(class)

    for key, value in pairs(props) do
        obj[key] = value
    end

    obj.Parent = parent
    return obj
end

local function Corner(obj, radius)
    Create("UICorner", {
        CornerRadius = UDim.new(0, radius or 8)
    }, obj)
end

local function Stroke(obj, color, transparency)
    Create("UIStroke", {
        Color = color or GOLD,
        Transparency = transparency or 0.3,
        Thickness = 1
    }, obj)
end

local function GetRoot(player)
    if not player or not player.Character then
        return nil
    end

    return player.Character:FindFirstChild("HumanoidRootPart")
end

local function GetHumanoid(player)
    if not player or not player.Character then
        return nil
    end

    return player.Character:FindFirstChildOfClass("Humanoid")
end

local function Alive(player)
    local hum = GetHumanoid(player)
    return hum and hum.Health > 0 and GetRoot(player)
end

--========================================================

local function GetTargets()
    local list = {}
    local myRoot = GetRoot(LP)

    if not myRoot then
        return list
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and Alive(player) then
            local root = GetRoot(player)

            if root then
                table.insert(list, {
                    Player = player,
                    Distance = (root.Position-myRoot.Position).Magnitude
                })
            end
        end
    end

    table.sort(list,function(a,b)
        return a.Distance < b.Distance
    end)

    return list
end

local function RemoveHighlight()
    if Highlight then
        Highlight:Destroy()
        Highlight = nil
    end
end

local function SetHighlight(player)
    RemoveHighlight()

    if not player or not player.Character then
        return
    end

    Highlight = Create("Highlight", {
        Adornee = player.Character,
        FillTransparency = 1,
        OutlineColor = GOLD,
        OutlineTransparency = 0,
        DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    }, workspace)
end

local function Unlock()
    Locked = false
    Target = nil

    RemoveHighlight()

    Camera = workspace.CurrentCamera

    if Camera then
        Camera.CameraType =
            OriginalCameraType or Enum.CameraType.Custom

        if OriginalCameraSubject and OriginalCameraSubject.Parent then
            Camera.CameraSubject = OriginalCameraSubject
        else
            local hum = GetHumanoid(LP)
            if hum then
                Camera.CameraSubject = hum
            end
        end
    end

    Status.Text = "STATUS | Lock On desativado"
    TargetInfo.Text = "ALVO | Nenhum"
end

local function LockPlayer(player)
    if not player or not Alive(player) then
        return
    end

    Camera = workspace.CurrentCamera

    if not Locked and Camera then
        OriginalCameraType = Camera.CameraType
        OriginalCameraSubject = Camera.CameraSubject
    end

    Target = player
    Locked = true

    SetHighlight(player)

    Status.Text = "STATUS | LOCK ON ATIVO"
    TargetInfo.Text = "ALVO | "..player.DisplayName
end

local function AutoTarget()
    local list = GetTargets()

    if #list == 0 then
        Status.Text = "STATUS | Nenhum alvo disponível"
        return
    end

    LockPlayer(list[1].Player)
end

local function CycleTarget(direction)
    local list = GetTargets()

    if #list == 0 then
        Status.Text = "STATUS | Nenhum alvo disponível"
        return
    end

    local index = 0

    for i, data in ipairs(list) do
        if data.Player == Target then
            index = i
            break
        end
    end

    if index == 0 then
        index = direction > 0 and 1 or #list
    else
        index = ((index-1+direction)%#list)+1
    end

    LockPlayer(list[index].Player)
end

Section("DOMINION AURA")

local function ClearAura()
    for _, obj in ipairs(AuraObjects) do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end

    table.clear(AuraObjects)
end

local function CreateAura()
    ClearAura()

    local character = LP.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if not root then return end

    local attachment = Create("Attachment", {
        Position = Vector3.new(0,-1.5,0)
    },root)

    table.insert(AuraObjects,attachment)

    local emitter = Create("ParticleEmitter", {
        Texture = "rbxasset://textures/particles/sparkles_main.dds",
        Color = ColorSequence.new(GOLD,LIGHT_GOLD),
        LightEmission = 0.8,
        Rate = 18 * CONFIG.AuraIntensity,
        Lifetime = NumberRange.new(0.6,1.2),
        Speed = NumberRange.new(0.5,1.3),
        SpreadAngle = Vector2.new(180,180),
        Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0,0.12),
            NumberSequenceKeypoint.new(0.5,0.22),
            NumberSequenceKeypoint.new(1,0)
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0,0.1),
            NumberSequenceKeypoint.new(1,1)
        })
    },attachment)

    table.insert(AuraObjects,emitter)

    local light = Create("PointLight", {
        Name = "NEXUS_AuraLight",
        Color = GOLD,
        Brightness = 0.6 * CONFIG.AuraIntensity,
        Range = 9
    },root)

    table.insert(AuraObjects,light)

    local ring = Create("Part", {
        Name = "NEXUS_AuraRing",
        Shape = Enum.PartType.Cylinder,
        Size = Vector3.new(0.07,5,5),
        CFrame = root.CFrame *
            CFrame.new(0,-2.5,0) *
            CFrame.Angles(0,0,math.rad(90)),
        Anchored = false,
        CanCollide = false,
        CanTouch = false,
        CanQuery = false,
        Massless = true,
        Material = Enum.Material.Neon,
        Color = GOLD,
        Transparency = 0.45
    },workspace)

    table.insert(AuraObjects,ring)

    Create("WeldConstraint", {
        Part0 = root,
        Part1 = ring
    },ring)
end

local function ApplyProfile(profile)
    CONFIG.Profile = profile

    if profile == "Balanced" then
        for effect,original in pairs(OriginalLighting) do
            if effect == "GlobalShadows" then
                Lighting.GlobalShadows = original
            elseif effect and effect.Parent then
                effect.Enabled = original
            end
        end

        Status.Text = "STATUS | Balanced aplicado"
        return
    end

    for _,effect in ipairs(Lighting:GetChildren()) do
        if effect:IsA("PostEffect") then
            if OriginalLighting[effect] == nil then
                OriginalLighting[effect] = effect.Enabled
            end

            effect.Enabled = false
        end
    end

    if OriginalLighting.GlobalShadows == nil then
        OriginalLighting.GlobalShadows = Lighting.GlobalShadows
    end

    if profile == "Ultra Low" then
        Lighting.GlobalShadows = false
    end

    Status.Text = "STATUS | "..profile.." aplicado"
end

Button("BALANCED",function()
    ApplyProfile("Balanced")
end)

    ApplyProfile("Performance")
end)

    ApplyProfile("Ultra Low")

local dragging = false
local dragStart
local startPosition

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position-dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset+delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset+delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

local renderConnection

renderConnection = RunService.RenderStepped:Connect(function(dt)
    frames += 1
    elapsed += dt

    if elapsed >= 1 then
        local fps = math.floor(frames/elapsed)
        frames = 0
        elapsed = 0

        local ping = "--"

        pcall(function()
            ping = Stats.Network.ServerStatsItem[
                "Data Ping"
            ]:GetValueString()
        end)

        Monitor.Text = "FPS "..fps.." | PING "..ping
    end

    if Locked then
        local root = GetRoot(Target)
        local hum = GetHumanoid(Target)
        local myRoot = GetRoot(LP)

        if not Target or not Target.Parent
        or not root or not hum or hum.Health <= 0
        or not myRoot then

            Unlock()
            return
        end

        Camera = workspace.CurrentCamera

        if Camera then
            Camera.CameraType = Enum.CameraType.Scriptable

            local focus = root.Position + Vector3.new(0,1.5,0)

            local offset =
                -root.CFrame.LookVector * CONFIG.Distance
                + Vector3.new(0,CONFIG.Height,0)

            local desired = CFrame.new(
                root.Position+offset,
                focus
            )

            local alpha = math.clamp(
                CONFIG.Smoothness*60*dt,
                0,
                1
            )

            Camera.CFrame = Camera.CFrame:Lerp(
                desired,
                alpha
            )

            local distance =
                (root.Position-myRoot.Position).Magnitude

            TargetInfo.Text = string.format(
                "ALVO | %s | %.1f studs",
                Target.DisplayName,
                distance
            )
        end
    end
end)
