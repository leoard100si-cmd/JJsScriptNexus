local function Create(class, props, parent)
    local obj = Instance.new(class)

    for key, value in pairs(props) do
        obj[key] = value
    end

    obj.Parent = parent
    return obj
end

local function Corner(obj, radius)
    Create("UICorner", {
        CornerRadius = UDim.new(0, radius or 8)
    }, obj)
end

local function Outline(obj, color, transparency)
    Create("UIStroke", {
        Color = color or Palette.Gold,
        Transparency = transparency or 0.3,
        Thickness = 1
    }, obj)
end

local function Label(parent, text, height, color)
    return Create("TextLabel", {
        Size = UDim2.new(1,-4,0,height or 28),
        BackgroundTransparency = 1,
        Text = text,
        TextColor3 = color or Palette.Muted,
        TextSize = 10,
        Font = Enum.Font.Gotham,
        TextWrapped = true,
        TextXAlignment = Enum.TextXAlignment.Left
    },parent)
end

    for pageName,page in pairs(Pages) do
        page.Visible = pageName == name
    end

    for pageName,button in pairs(Tabs) do
        local selected = pageName == name

            and Color3.fromRGB(70,55,30)
            or Palette.Panel

        button.TextColor3 = selected
            and Palette.LightGold
            or Palette.Text
    end
end

--========================================================
-- CAMERA MANAGEMENT
--========================================================

local function CaptureCamera()
    local camera = workspace.CurrentCamera

    if camera and Original.FOV == nil then
        Original.FOV = camera.FieldOfView
    end
end

local function SetFOV(value)
    C.FOV = value

    local camera = workspace.CurrentCamera

    if camera then
        camera.FieldOfView = value
    end

    if Status then
        Status.Text = "SYSTEM | FOV " .. value
    end
end

local function RestoreCamera()
    local camera = workspace.CurrentCamera

    if camera and Original.FOV then
        camera.FieldOfView = Original.FOV
    end
end

--========================================================
-- AURA SYSTEM
--========================================================

local function ClearAura()
    for _,obj in ipairs(AuraObjects) do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end

    table.clear(AuraObjects)
end

local function CreateAura()
    ClearAura()

    local character = LP.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")

    if not root then
        return
    end

    local colors = AuraColors[C.AuraStyle]

    local attachment = Create("Attachment", {
        Position = Vector3.new(0,-1.5,0)
    },root)

    table.insert(AuraObjects,attachment)

    local emitter = Create("ParticleEmitter", {
        Texture = "rbxasset://textures/particles/sparkles_main.dds",
        Color = ColorSequence.new(colors[1],colors[2]),
        LightEmission = 0.8,
        Rate = 18 * C.AuraIntensity,
        Lifetime = NumberRange.new(0.5,1.1),
        Speed = NumberRange.new(0.5,1.3),
        SpreadAngle = Vector2.new(180,180),
        Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0,0.16),
            NumberSequenceKeypoint.new(1,0)
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0,0.1),
            NumberSequenceKeypoint.new(1,1)
        })
    },attachment)

    table.insert(AuraObjects,emitter)

    local light = Create("PointLight", {
        Color = colors[1],
        Brightness = 0.5 * C.AuraIntensity,
        Range = 9
    },root)

    table.insert(AuraObjects,light)
end

local function CaptureLighting()
    if Original.GlobalShadows == nil then
        Original.GlobalShadows = Lighting.GlobalShadows
    end

    for _,effect in ipairs(Lighting:GetChildren()) do
        if effect:IsA("PostEffect")
        and Original.Effects[effect] == nil then
            Original.Effects[effect] = effect.Enabled
        end
    end
end

local function RestoreLighting()
    if Original.GlobalShadows ~= nil then
        Lighting.GlobalShadows = Original.GlobalShadows
    end

    for effect,enabled in pairs(Original.Effects) do
        if effect and effect.Parent then
            effect.Enabled = enabled
        end
    end
end

local function ApplyPerformance(mode)
    CaptureLighting()

    C.Performance = mode

    if mode == 1 then
        RestoreLighting()
        Status.Text = "SYSTEM | BALANCED"
    elseif mode == 2 then
        Lighting.GlobalShadows = false
        Status.Text = "SYSTEM | PERFORMANCE"
    elseif mode == 3 then
        Lighting.GlobalShadows = false

        for effect,_ in pairs(Original.Effects) do
            if effect and effect.Parent then
                effect.Enabled = false
            end
        end

        Status.Text = "SYSTEM | ULTRA LOW"
    end
end

local function SetGodOn(enabled)
    C.Active = enabled

    if enabled then
        CaptureCamera()
        CaptureLighting()

        if C.Aura then
            CreateAura()
        end

        SetFOV(C.FOV)
        Status.Text = "SYSTEM | GOD ON ACTIVE"
    else
        ClearAura()
        RestoreCamera()
        RestoreLighting()

        Status.Text = "SYSTEM | STANDBY"
    end

    GodButton.Text = enabled
        and "GOD ON | ACTIVE"
        or "GOD ON | OFF"

    GodButton.BackgroundColor3 = enabled
        and Color3.fromRGB(75,55,25)
        or Palette.Panel

    Floating.Text = enabled and "ON" or "N"
    Floating.TextColor3 = enabled
        and Palette.Green
        or Palette.LightGold
end

        local fps = math.floor(frames / elapsed)

        frames = 0
        elapsed = 0

        pcall(function()
            ping = Stats.Network.ServerStatsItem[
                "Data Ping"
            ]:GetValueString()
        end)

        Monitor.Text = "FPS " .. fps .. " | PING " .. ping
    end
end))

--========================================================
-- DRAG SUPPORT
--========================================================

local dragging = false
local dragStart
local startPosition
local dragInput

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then

        dragging = true
        dragInput = input
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

Track(UIS.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseMovement
    ) then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end))

Track(UIS.InputEnded:Connect(function(input)
    if input == dragInput then
        dragging = false
        dragInput = nil
    end
end))
